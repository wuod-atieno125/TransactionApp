async function handle(fn, req, res, next) {
  try {
    return res.status(200).json({ success: true, ...(await fn()) });
  } catch (error) {
    return next(error);
  }
}

function getPagination(queryParams) {
  const parseIntegerParam = (value) => {
    if (value === undefined || value === "") {
      return null;
    }

    if (
      typeof value !== "string" ||
      !/^\d+$/.test(value)
    ) {
      return NaN;
    }

    const parsed = Number(value);

    return Number.isSafeInteger(parsed) ? parsed : NaN;
  };

  const parsedLimit = parseIntegerParam(queryParams.limit);
  const parsedOffset = parseIntegerParam(queryParams.offset);

  if (
    Number.isNaN(parsedLimit) ||
    Number.isNaN(parsedOffset) ||
    (parsedLimit !== null && parsedLimit < 1) ||
    (parsedOffset !== null && parsedOffset < 0)
  ) {
    return { error: "Invalid pagination parameters." };
  }

  return {
    limit: parsedLimit === null ? 50 : Math.min(parsedLimit, 100),
    offset: parsedOffset === null ? 0 : parsedOffset,
  };
}
function isValidDateRange(startDate, endDate, maxDays = 366) {
  const datePattern = /^\d{4}-\d{2}-\d{2}$/;

  if (
    typeof startDate !== "string" ||
    typeof endDate !== "string" ||
    !datePattern.test(startDate) ||
    !datePattern.test(endDate)
  ) {
    return false;
  }

  function toUtcDay(value) {
    const [year, month, day] = value.split("-").map(Number);
    const date = new Date(Date.UTC(year, month - 1, day));

    if (
      date.getUTCFullYear() !== year ||
      date.getUTCMonth() !== month - 1 ||
      date.getUTCDate() !== day
    ) {
      return null;
    }

    return Math.floor(date.getTime() / 86400000);
  }

  const start = toUtcDay(startDate);
  const end = toUtcDay(endDate);

  return (
    start !== null &&
    end !== null &&
    end >= start &&
    end - start < maxDays
  );
}

async function reverseTransaction(client, {
  transaction,
  actorUserId,
  reason,
  approvalRequestId = null,
  excludedMemberUserId = null,
}) {
  const result = await client.query(
    `UPDATE transactions
     SET status = 'reversed',
         updated_by = $1,
         updated_at = NOW()
     WHERE id = $2
       AND status = 'active'
     RETURNING *`,
    [actorUserId, transaction.id]
  );
  const reversedTransaction = result.rows[0];

  if (!reversedTransaction) {
    const error = new Error("Active transaction not found.");
    error.statusCode = 404;
    throw error;
  }

  await client.query(
    `INSERT INTO audit_logs (
       actor_user_id, action, target_type, target_id,
       old_values, new_values
     )
     VALUES ($1, 'transaction_reversed', 'transaction', $2, $3::jsonb, $4::jsonb)`,
    [
      actorUserId,
      transaction.id,
      JSON.stringify(transaction),
      JSON.stringify({
        ...reversedTransaction,
        reversal_reason: reason,
        reversal_approval_request_id: approvalRequestId,
      }),
    ]
  );

  await client.query(
    `INSERT INTO notifications (
       recipient_user_id, transaction_id, event_type, title, message
     )
     SELECT id, $1, 'transaction_updated',
            'Contribution reversal completed',
            'A contribution reversal was approved and completed.'
     FROM users
     WHERE role = 'member'
       AND account_status = 'approved'
       AND deleted_at IS NULL
       AND ($2::uuid IS NULL OR id <> $2)`,
    [
      transaction.id,
      excludedMemberUserId,
    ]
  );

  return reversedTransaction;
}

router.get(
  "/",
  requireAuth,
  requireRole("member", "treasurer", "chairperson", "secretary"),
  async (req, res, next) => {
    try {
      const pagination = getPagination(req.query);

      if (pagination.error) {
        return res.status(400).json({
          success: false,
          message: pagination.error,
        });
      }

      const { limit, offset } = pagination;

      const role = String(req.session.user.role || "").toLowerCase();

      let result;

      if (role === "member" || role === "secretary") {
        result = await query(
          `SELECT transactions.id, transactions.member_user_id,
                  transactions.entered_member_name, transactions.amount,
                  transactions.paid_at, transactions.mpesa_reference,
                  transactions.status, transactions.notes,
                  transactions.created_at, transactions.updated_at,
                  reversal_request.status AS reversal_request_status,
                  transactions.created_at >= NOW() - INTERVAL '12 hours'
                    AS reversal_allowed
           FROM transactions
           LEFT JOIN LATERAL (
             SELECT status
             FROM transaction_reversal_requests
             WHERE transaction_id = transactions.id
             ORDER BY created_at DESC
             LIMIT 1
           ) reversal_request ON TRUE
           WHERE transactions.member_user_id = $1
             AND (
               transactions.status = 'active'
               OR reversal_request.status = 'approved'
             )
           ORDER BY transactions.paid_at DESC
           LIMIT $2 OFFSET $3`,
          [req.session.user.id, limit, offset]
        );
      } else {
        result = await query(
          `SELECT transactions.id, transactions.member_user_id,
                  transactions.entered_member_name, transactions.amount,
                  transactions.paid_at, transactions.mpesa_reference,
                  transactions.status, transactions.notes,
                  transactions.created_at, transactions.updated_at,
                  reversal_request.status AS reversal_request_status,
                  transactions.created_at >= NOW() - INTERVAL '12 hours'
                    AS reversal_allowed
           FROM transactions
           LEFT JOIN LATERAL (
             SELECT status
             FROM transaction_reversal_requests
             WHERE transaction_id = transactions.id
             ORDER BY created_at DESC
             LIMIT 1
           ) reversal_request ON TRUE
           WHERE transactions.status IN ('active', 'reversed')
           ORDER BY transactions.paid_at DESC
           LIMIT $1 OFFSET $2`,
          [limit, offset]
        );
      }

      return res.status(200).json({
        success: true,
        transactions: result.rows,
        pagination: {
          limit,
          offset,
          count: result.rows.length,
        },
      });
    } catch (error) {
      next(error);
    }
  }

         FROM transactions
         WHERE id = $1 AND status = 'active'
         FOR UPDATE`,
        [id]
      );

      if (existingResult.rowCount === 0) {
        await client.query("ROLLBACK").catch(()=>{});
        return res.status(404).json({
          success: false,
          message: "Active transaction not found.",
        });
      }

      const oldTransaction = existingResult.rows[0];

      const name =
        enteredMemberName === undefined
          ? oldTransaction.entered_member_name
          : enteredMemberName.trim();

      const newAmount =
        amount === undefined ? oldTransaction.amount : Number(amount);

      const newPaidAt =
        paidAt === undefined ? oldTransaction.paid_at : paidAt;

      const reference =
        mpesaReference === undefined
          ? oldTransaction.mpesa_reference
          : typeof mpesaReference === "string" && mpesaReference.trim()
            ? mpesaReference.trim()
            : null;

      const normalizedReference = reference
        ? reference.toUpperCase()
        : null;

      const newNotes =
        notes === undefined
          ? oldTransaction.notes
          : typeof notes === "string" && notes.trim()
            ? notes.trim()
            : null;

      const updatedResult = await client.query(
        `UPDATE transactions
         SET entered_member_name = $1,
             amount = $2,
             paid_at = $3,
             mpesa_reference = $4,
             normalized_mpesa_reference = $5,
             notes = $6,
             updated_by = $7,
             updated_at = NOW()
         WHERE id = $8
         RETURNING *`,
        [
          name,
          newAmount,
          newPaidAt,
          reference,
          normalizedReference,
          newNotes,
          req.session.user.id,
          id,
        ]
      );

      const updatedTransaction = updatedResult.rows[0];

      await client.query(
        `INSERT INTO audit_logs (
           actor_user_id, action, target_type, target_id,
           old_values, new_values
         )
         VALUES ($1, $2, $3, $4, $5::jsonb, $6::jsonb)`,
        [
          req.session.user.id,
          "transaction_updated",
          "transaction",
          id,
          JSON.stringify(oldTransaction),
          JSON.stringify({
            ...updatedTransaction,
            change_reason: reason,
          }),
        ]
      );

      if (updatedTransaction.member_user_id) {
        await client.query(
          `INSERT INTO notifications (
             recipient_user_id, transaction_id,
             event_type, title, message
           )
           VALUES ($1, $2, $3, $4, $5)`,
          [
            updatedTransaction.member_user_id,
            id,
            "transaction_updated",
            "Contribution updated",
            `Your contribution of ${updatedTransaction.amount} has been updated.`,
          ]
        );
      }

      await client.query("COMMIT");

      return res.status(200).json({
        success: true,
        message: "Transaction updated successfully.",
        transaction: updatedTransaction,
      });
    } catch (error) {

      if (client) {
        await client.query("ROLLBACK").catch((rollbackError) => {
          console.error("Transaction update rollback failed:", rollbackError);
        });
      }

      if (error.code === "23505") {
        return res.status(409).json({
          success: false,
          message: "This M-Pesa reference has already been recorded.",
        });
      }

      return next(error);
    } finally {
      if (client) {
        client.release();
      }
    }
  }

function parsePagination(queryParams = {}) {
  const parse = (value, fallback) => {
    if (value === undefined || value === "") return fallback;
    if (typeof value !== "string" || !/^\d+$/.test(value)) {
      const error = new Error("Invalid pagination parameters.");
      error.statusCode = 400;
      throw error;
    }

    const parsed = Number(value);
    if (!Number.isSafeInteger(parsed)) {
      const error = new Error("Invalid pagination parameters.");
      error.statusCode = 400;
      throw error;
    }

    return parsed;
  };

  const limit = parse(queryParams.limit, 50);
  const offset = parse(queryParams.offset, 0);

  if (limit < 1 || offset < 0) {
    const error = new Error("Invalid pagination parameters.");
    error.statusCode = 400;
    throw error;
  }

  return {
    limit: Math.min(limit, 100),
    offset,
  };
}

async function getContribution(id, actor) {
  return repository.findById(id, actor);
}

async function listContributions(actor, queryParams) {
  const pagination = parsePagination(queryParams);
  const contributions = await repository.list({ ...actor, ...pagination });

  return { contributions, pagination };
}

function assertAmount(value, field = "Amount") {
  const amount = typeof value === "number" || typeof value === "string"
    ? Number(value)
    : NaN;

  if (!Number.isFinite(amount) || amount <= 0 || amount > 9999999999.99
      || !/^\d+(\.\d{1,2})?$/.test(String(value))) {
    const error = new Error(`${field} must be positive with no more than two decimal places.`);
    error.statusCode = 400;
    throw error;
  }

  return amount.toFixed(2);
}

function requireText(value, field, maxLength) {
  if (typeof value !== "string" || value.trim().length < 3 || value.trim().length > maxLength) {
    const error = new Error(`${field} must be between 3 and ${maxLength} characters.`);
    error.statusCode = 400;
    throw error;
  }
  return value.trim();
}

async function recordContribution({ memberUserId, amount, description, userId }) {
  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(memberUserId || "")) {

      `SELECT id, full_name FROM users
       WHERE id = $1 AND role = 'member' AND account_status = 'approved' AND deleted_at IS NULL`,

      error.statusCode = 400;
      throw error;

      const error = new Error("Contribution amount must match the currently required welfare amount.");
      error.statusCode = 400;
      throw error;
    }
    await client.query("SELECT pg_advisory_xact_lock(284731)");

async function getStatement(memberUserId) {
  const result = await query(
    `SELECT u.id, u.full_name, u.phone,
            COALESCE(SUM(CASE WHEN l.entry_type = 'deposit' THEN l.amount ELSE -l.amount END), 0)::numeric(12,2) AS balance,
            COALESCE(SUM(l.amount) FILTER (WHERE l.entry_type = 'deposit'), 0)::numeric(12,2) AS total_contributions,
            json_agg(l ORDER BY l.created_at DESC) FILTER (WHERE l.id IS NOT NULL) AS entries
     FROM users u
     LEFT JOIN welfare_ledger_entries l ON l.member_user_id = u.id
     WHERE u.id = $1 AND u.role = 'member' AND u.deleted_at IS NULL
     GROUP BY u.id`,
    [memberUserId]
  );
  return result.rows[0] || null;
}

async function executeWithdrawal({ requestId, userId }) {
  const client = await getClient();
  try {
    await client.query("BEGIN");
    await client.query("SELECT pg_advisory_xact_lock(284731)");
    const request = await client.query(
      `SELECT * FROM welfare_withdrawal_requests WHERE id = $1 FOR UPDATE`,
      [requestId]
    );
    if (!request.rowCount || request.rows[0].status !== "approved") {
      const error = new Error("Only an approved welfare withdrawal can be executed.");
      error.statusCode = 409;
      throw error;
    }
    const members = await client.query(
      `SELECT u.id, u.full_name, u.phone,
              COALESCE(SUM(CASE WHEN l.entry_type = 'deposit' THEN l.amount ELSE -l.amount END), 0)::numeric(12,2) AS balance
       FROM users u LEFT JOIN welfare_ledger_entries l ON l.member_user_id = u.id
       WHERE u.role = 'member' AND u.account_status = 'approved' AND u.deleted_at IS NULL
       GROUP BY u.id FOR UPDATE OF u`
    );
    const total = members.rows.reduce((sum, member) => sum + Number(member.balance), 0);
    const deductionTotal = members.rows.reduce((sum, member) => sum + Number((Number(member.balance) / 2).toFixed(2)), 0);
    if (Number(request.rows[0].requested_amount) !== Number(deductionTotal.toFixed(2))) {
      const error = new Error("Requested amount must equal 50% of the current welfare balance.");
      error.statusCode = 409;
      throw error;
    }
    for (const member of members.rows) {
      const before = Number(member.balance);
      const amount = Number((before / 2).toFixed(2));
      if (!amount) continue;
      await client.query(
        `INSERT INTO welfare_ledger_entries
         (member_user_id, entry_type, amount, balance_before, balance_after,
          welfare_withdrawal_request_id, recorded_by, description)
         VALUES ($1, 'deduction', $2, $3, $4, $5, $6, $7)`,
        [member.id, amount.toFixed(2), before.toFixed(2), (before - amount).toFixed(2),
         requestId, userId, "50% collective welfare withdrawal deduction"]
      );
    }
    await client.query(
      `INSERT INTO welfare_ledger_entries
       (entry_type, amount, balance_before, balance_after, welfare_withdrawal_request_id, recorded_by, description)
       VALUES ('withdrawal', $1, $2, $3, $4, $5, $6)`,
      [deductionTotal.toFixed(2), total.toFixed(2), (total - deductionTotal).toFixed(2),
       requestId, userId, request.rows[0].purpose]
    );
    const updated = await client.query(
      `UPDATE welfare_withdrawal_requests
       SET status = 'executed', executed_by = $1, executed_at = NOW()
       WHERE id = $2 RETURNING *`,
      [userId, requestId]
    );
    await client.query(
      `INSERT INTO notifications (recipient_user_id, event_type, title, message)
       SELECT id, 'welfare_withdrawal_executed', 'Collective welfare withdrawal executed',
              $1 FROM users WHERE role = 'member' AND account_status = 'approved'
              AND deleted_at IS NULL`,
      [`KSh ${deductionTotal.toFixed(2)} was deducted at 50% from each member's welfare balance.`]
    );
    await client.query("COMMIT");
    return updated.rows[0];
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
}
